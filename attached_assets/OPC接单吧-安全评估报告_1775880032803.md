### V-01 ｜ P0 严重 ｜ CORS 策略完全开放

**现象**

```
HTTP Response Headers:
access-control-allow-origin: *
access-control-allow-methods: GET,HEAD,PUT,PATCH,POST,DELETE
```

所有来源的跨域请求均被允许，且支持全部 HTTP 方法。

**攻击场景**

1. 攻击者创建恶意网站 `evil-site.com`
2. 已登录的 opcorder.com 用户访问该恶意网站
3. 恶意 JS 直接调用 `https://www.opcorder.com/api/users/me` 等接口
4. 由于 `Access-Control-Allow-Origin: *`，浏览器不拦截，攻击者获取用户数据

**影响范围**：所有已登录用户的数据可被第三方站点窃取。

**整改策略**

```javascript
// server/middleware/cors.js
const cors = require('cors');

const ALLOWED_ORIGINS = [
  'https://www.opcorder.com',
  'https://opcorder.com',
  // 开发环境按需添加
  ...(process.env.NODE_ENV === 'development' ? ['http://localhost:5173'] : [])
];

const corsOptions = {
  origin: (origin, callback) => {
    // 允许无 origin 的请求（如移动端 App、curl）
    if (!origin || ALLOWED_ORIGINS.includes(origin)) {
      callback(null, true);
    } else {
      callback(new Error('Not allowed by CORS'));
    }
  },
  methods: ['GET', 'POST', 'PUT', 'PATCH', 'DELETE'],
  credentials: true,          // 允许携带 Cookie
  maxAge: 86400,              // 预检缓存 24h
  allowedHeaders: ['Content-Type', 'Authorization'],
};

module.exports = cors(corsOptions);
```

**验证方法**：

```bash
# 用恶意 Origin 测试，应返回无 CORS 头或被拒绝
curl -sI https://www.opcorder.com/api/posts \
  -H "Origin: https://evil.com" | grep access-control
# 预期：无输出或返回错误
```

### V-02 ｜ P0 严重 ｜ 注册接口无邮箱验证与人机校验

**现象**

```bash
curl -X POST https://www.opcorder.com/api/auth/register \
  -H "Content-Type: application/json" \
  -d '{"email":"任意邮箱@example.com","password":"Test123456!","nickname":"Test","role":"opc"}'

# 直接返回成功，账号立即可用：
# {"id":340,"nickname":"SecurityTest","email":"test_sec_audit@example.com","role":"opc",...}
```

无邮箱验证、无 CAPTCHA、无频率限制。

**攻击场景**

- 脚本批量注册数万个垃圾账号
- 利用垃圾账号发布广告、诈骗信息
- 注册接口被用于邮箱有效性探测

**整改策略（分三层防御）**

**第一层：CAPTCHA 人机校验（立即实施）**

推荐使用 Cloudflare Turnstile（免费、对用户友好）：

```javascript
// server/middleware/captcha.js
async function verifyCaptcha(req, res, next) {
  const token = req.body.captchaToken;
  if (!token) {
    return res.status(400).json({ error: '请完成人机验证' });
  }

  const response = await fetch(
    'https://challenges.cloudflare.com/turnstile/v0/siteverify',
    {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        secret: process.env.TURNSTILE_SECRET_KEY,
        response: token,
      }),
    }
  );
  const data = await response.json();

  if (!data.success) {
    return res.status(403).json({ error: '人机验证失败，请重试' });
  }
  next();
}

// 应用到注册和登录路由
app.post('/api/auth/register', verifyCaptcha, registerHandler);
app.post('/api/auth/login', verifyCaptcha, loginHandler);
```

**第二层：邮箱验证流程**

```
注册请求 → 创建"待验证"账号 → 发送验证邮件 → 用户点击链接 → 账号激活
```

```javascript
// 注册时标记为未验证
const user = await db.insert('users').values({
  ...userData,
  emailVerified: false,
  verificationToken: crypto.randomUUID(),
  verificationExpiry: new Date(Date.now() + 24 * 60 * 60 * 1000) // 24h
});

// 发送验证邮件
await sendEmail({
  to: user.email,
  subject: '接单吧 - 请验证您的邮箱',
  html: `请点击链接完成验证：
    https://www.opcorder.com/verify-email?token=${user.verificationToken}`
});

// 登录时检查
if (!user.emailVerified) {
  return res.status(403).json({ error: '请先验证邮箱' });
}
```

**第三层：注册频率限制**

```javascript
const rateLimit = require('express-rate-limit');

const registerLimiter = rateLimit({
  windowMs: 60 * 60 * 1000,  // 1小时窗口
  max: 3,                     // 每IP最多3次注册
  message: { error: '注册过于频繁，请稍后再试' },
  standardHeaders: true,
  legacyHeaders: false,
});

app.post('/api/auth/register', registerLimiter, verifyCaptcha, registerHandler);
```

### V-03 ｜ P1 高危 ｜ 登录接口无速率限制

**现象**

连续 5 次发送错误密码的登录请求，全部正常返回 `{"error":"账号或密码错误"}`，无任何限制。

**攻击场景**：攻击者用字典对已知邮箱暴力破解密码，每秒可尝试数百次。

**整改策略（双维度限制）**

```javascript
// === 维度 1：基于 IP 的全局限制 ===
const loginIpLimiter = rateLimit({
  windowMs: 15 * 60 * 1000,   // 15分钟
  max: 20,                      // 每IP最多20次登录尝试
  message: { error: '登录尝试过于频繁，请15分钟后重试' },
  keyGenerator: (req) => req.ip,
});

// === 维度 2：基于账号的限制（需要自定义实现）===
const loginAttempts = new Map(); // 生产环境建议用 Redis

async function accountLockout(req, res, next) {
  const { email } = req.body;
  const key = `login:${email}`;
  const attempts = loginAttempts.get(key) || { count: 0, lockedUntil: null };

  // 检查是否被锁定
  if (attempts.lockedUntil && attempts.lockedUntil > Date.now()) {
    const minutesLeft = Math.ceil((attempts.lockedUntil - Date.now()) / 60000);
    return res.status(429).json({
      error: `账号已被临时锁定，请${minutesLeft}分钟后重试`
    });
  }

  // 失败5次后要求 CAPTCHA，失败10次后锁定15分钟
  if (attempts.count >= 10) {
    attempts.lockedUntil = Date.now() + 15 * 60 * 1000;
    attempts.count = 0;
    loginAttempts.set(key, attempts);
    return res.status(429).json({ error: '登录失败次数过多，账号已被锁定15分钟' });
  }

  if (attempts.count >= 5) {
    // 要求 CAPTCHA（结合 V-02 的 CAPTCHA 验证）
    req.requireCaptcha = true;
  }

  req.loginAttempts = attempts;
  next();
}

// 登录成功后重置计数
function resetLoginAttempts(email) {
  loginAttempts.delete(`login:${email}`);
}

app.post('/api/auth/login', loginIpLimiter, accountLockout, loginHandler);
```

### V-04 ｜ P1 高危 ｜ 密码重置接口存在用户枚举漏洞

**现象**

```bash
# 不存在的邮箱
curl -X POST /api/auth/forgot-password -d '{"email":"nonexistent@test.com"}'
# → {"error":"该邮箱未注册，请检查邮箱地址是否正确"}

# 攻击者可通过不同的错误信息判断邮箱是否注册
```

**整改策略**

```javascript
app.post('/api/auth/forgot-password', async (req, res) => {
  const { email } = req.body;

  // 无论邮箱是否存在，都返回相同响应
  const UNIFIED_RESPONSE = {
    message: '如果该邮箱已注册，我们已向其发送了密码重置链接，请查收邮件'
  };

  try {
    const user = await findUserByEmail(email);

    if (user) {
      // 真正发送重置邮件
      const token = crypto.randomUUID();
      await saveResetToken(user.id, token, Date.now() + 30 * 60 * 1000);
      await sendResetEmail(email, token);
    }
    // 不管找没找到用户，都返回同样的消息
  } catch (err) {
    // 即使出错也返回同样消息，避免通过响应时间差异推断
  }

  // 统一延迟响应（防止时序攻击）
  await new Promise(resolve => setTimeout(resolve, 500 + Math.random() * 500));
  return res.json(UNIFIED_RESPONSE);
});
```

### V-05 ｜ P1 高危 ｜ 敏感业务数据 API 未鉴权

**现象**

以下接口无需任何认证即可访问：

| 端点                          | 泄露的敏感数据                                        |
| ----------------------------- | ----------------------------------------------------- |
| `GET /api/stats/overview`     | 平台总支付金额 (¥334,000)、活跃OPC数 (311)、月订单数 (7)、完成率 (28.57%) |

| `GET /api/posts`              | 社区帖子含 authorId、authorName                       |

**整改策略**

```javascript
// === 1. /api/stats/overview 必须鉴权 ===
app.get('/api/stats/overview', requireAuth, requireRole('admin'), statsHandler);

// === 2. 公开接口脱敏处理 ===
// 排行榜：移除内部 ID，用公开标识符替代
app.get('/api/users/opc-leaderboard', async (req, res) => {

    avatar: u.avatar,
    level: u.level,
    bio: u.bio?.substring(0, 100),  // 截断
    // 不返回 email、phone 等
  }));
  res.json(sanitized);
});

// === 3. 帖子接口：移除 authorId ===
// 只返回 authorName，不返回内部 authorId
```

### V-06 ｜ P2 中危 ｜ 缺少关键 HTTP 安全响应头

**现象**

| 安全头                    | 当前状态 | 作用                        |
| ------------------------- | -------- | --------------------------- |
| Content-Security-Policy   | 缺失     | 防止 XSS、数据注入          |
| X-Frame-Options           | 缺失     | 防止点击劫持                 |
| X-Content-Type-Options    | 缺失     | 防止 MIME 类型嗅探           |
| Referrer-Policy           | 缺失     | 控制来源信息泄露             |
| Permissions-Policy        | 缺失     | 限制浏览器功能（摄像头等）    |

**整改策略**

```bash
npm install helmet
```

```javascript
const helmet = require('helmet');

app.use(helmet({
  contentSecurityPolicy: {
    directives: {
      defaultSrc: ["'self'"],
      scriptSrc: ["'self'"],
      styleSrc: ["'self'", "'unsafe-inline'", "https://fonts.googleapis.com"],
      fontSrc: ["'self'", "https://fonts.gstatic.com"],
      imgSrc: ["'self'", "https://images.unsplash.com", "https://api.dicebear.com", "data:"],
      connectSrc: ["'self'"],
      frameAncestors: ["'none'"],
    }
  },
  crossOriginEmbedderPolicy: false,  // 如需加载外部图片
}));
```

### V-07 ｜ P2 中危 ｜ Cookie 缺少安全属性

**现象**

```
Set-Cookie: GAESA=CpwB...; expires=Fri, 11-May-2026; path=/
```

仅设置了 `expires` 和 `path`，缺少 `HttpOnly`、`Secure`、`SameSite`。

**整改策略**

GAESA 是 GCP 自动管理的会话 Cookie，无法直接修改。但应用自身的会话 Cookie 需确保：

```javascript
app.use(session({
  cookie: {
    httpOnly: true,        // JS 无法读取，防 XSS 窃取
    secure: true,          // 仅 HTTPS 传输
    sameSite: 'strict',    // 防 CSRF
    maxAge: 24 * 60 * 60 * 1000,  // 24 小时
    path: '/',
  },
  // ...
}));
```

如使用 JWT，建议存储在 `HttpOnly` Cookie 中，而非 `localStorage`。

### V-08 ｜ P2 中危 ｜ 服务器信息泄露

**现象**

| 泄露点                            | 泄露内容                           |
| --------------------------------- | ---------------------------------- |
| `Server: Google Frontend`         | 暴露托管在 Google Cloud             |
| `X-Powered-By: Express`          | 暴露后端框架为 Express.js           |
| `X-Cloud-Trace-Context: 77ad...` | 暴露 GCP 内部追踪 ID               |
| 默认错误页                        | `Cannot POST /api/auth/reset-password` 暴露路由和框架 |

**整改策略**

```javascript
// 1. 移除 X-Powered-By
app.disable('x-powered-by');

// 2. 自定义 404 处理器
app.use((req, res) => {
  res.status(404).json({ error: '请求的资源不存在' });
});

// 3. 自定义全局错误处理器
app.use((err, req, res, next) => {
  // 生产环境不输出堆栈
  console.error(err);  // 仅服务端日志
  res.status(err.status || 500).json({
    error: process.env.NODE_ENV === 'production'
      ? '服务器内部错误'
      : err.message
  });
});
```

GCP 的 `Server` 和 `X-Cloud-Trace-Context` 头在 App Engine 中无法直接移除，可在前面加一层 Cloudflare 代理来隐藏。

## 六、后续建议

### 1. 测试范围之外，但强烈建议关注的安全领域

| 领域                     | 建议                                                          |
| ------------------------ | ------------------------------------------------------------- |
| 身份认证（已登录状态）    | JWT/Session 安全性、Token 过期策略、刷新机制                    |
| 权限控制（RBAC）          | OPC/发单方/管理员 之间的水平越权测试                            |
| 文件上传                  | 上传类型白名单、文件大小限制、防止恶意文件执行                   |
| 支付/资金相关             | 金额篡改、订单状态篡改、并发下单、退款逻辑                       |
| 数据库安全                | SQL 注入深度测试（POST body、Header 注入）、ORM 配置            |
| 日志与监控                | 部署 WAF（如 Cloudflare）、异常登录告警、API 调用审计            |
| 依赖安全                  | 定期 `npm audit`，CI/CD 中集成 Snyk 或 Dependabot              |
| 备份与恢复                | 数据库定期备份、灾难恢复演练                                    |

### 2. 推荐的安全工具链

```
开发阶段 ──→ ESLint security plugin + npm audit
```
